// 批量操作优化
public class StudentDao {
    
    // 单条插入 vs 批量插入
    public void insertStudentsOneByOne(List<Student> students) {
        for (Student student : students) {
            insert(student);  // N次数据库调用
        }
    }
    
    public void insertStudentsBatch(List<Student> students) {
        String sql = "INSERT INTO students (id, name, class) VALUES (?, ?, ?)";
        try (Connection conn = getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            
            conn.setAutoCommit(false);  // 关闭自动提交
            
            for (Student student : students) {
                stmt.setString(1, student.getId());
                stmt.setString(2, student.getName());
                stmt.setString(3, student.getClassName());
                stmt.addBatch();  // 添加到批处理
            }
            
            stmt.executeBatch();  // 一次性执行
            conn.commit();       // 提交事务
        }
    }
}
